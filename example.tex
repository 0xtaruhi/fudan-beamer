\begin{frame}{定义与定理}
  \key{定理环境自动编号，名称为中文}
  \small
  \begin{definition}[凸函数]
    若对任意 $x, y \in \mathbb{R}^n$ 与 $\theta \in [0,1]$，有
    $f(\theta x + (1-\theta) y) \le \theta f(x) + (1-\theta) f(y)$，则称 $f$ 为凸函数。
  \end{definition}
  \begin{theorem}[Jensen 不等式]
    若 $f$ 为凸函数，$X$ 为随机变量，则 $f(\mathbb{E}[X]) \le \mathbb{E}[f(X)]$。
  \end{theorem}
\end{frame}

\begin{frame}{引理与证明}
  \key{proof 环境以强调色方块结尾}
  \footnotesize
  \begin{lemma}
    对任意正整数 $n$，有 $\sum_{k=1}^{n} k = n(n+1)/2$。
  \end{lemma}
  \begin{proof}
    对 $n$ 归纳。$n = 1$ 时两边均为 $1$。若结论对 $n$ 成立，则
    $\sum_{k=1}^{n+1} k = \frac{n(n+1)}{2} + (n+1) = \frac{(n+1)(n+2)}{2}$，即结论对 $n+1$ 也成立。
  \end{proof}
\end{frame}

def fib(n):
    """返回第 n 个斐波那契数。"""
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

print([fib(i) for i in range(10)])
\end{lstlisting}

long sum(const int *a, int n) {
    long s = 0;
    for (int i = 0; i < n; i++)
        s += a[i];
    return s;
}
\end{lstlisting}

\begin{frame}{算法}
  \key{algorithm2e 排版伪代码}
  \footnotesize
  \begin{algorithm}[H]
    \caption{二分查找}
    \KwIn{有序数组 $a[1..n]$，目标值 $x$}
    \KwOut{$x$ 的下标；若不存在则返回 $0$}
    $l \gets 1$,\ $r \gets n$\;
    \While{$l \le r$}{
      $m \gets \lfloor (l + r) / 2 \rfloor$\;
      \uIf{$a[m] = x$}{\Return $m$\;}
      \uElseIf{$a[m] < x$}{$l \gets m + 1$\;}
      \Else{$r \gets m - 1$\;}
    }
    \Return $0$\;
  \end{algorithm}
\end{frame}
